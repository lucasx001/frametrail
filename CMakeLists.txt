cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/version.props" FRAMETRAIL_VERSION_XML)
string(REGEX MATCH "<Version>([0-9]+\\.[0-9]+\\.[0-9]+)</Version>" FRAMETRAIL_VERSION_MATCH "${FRAMETRAIL_VERSION_XML}")
if(NOT FRAMETRAIL_VERSION_MATCH)
  message(FATAL_ERROR "version.props must contain a three-component Version.")
endif()
project(FrameTrail VERSION ${CMAKE_MATCH_1} LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "FrameTrail requires Windows; use the Windows x64 toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Each native DLL owns its allocations; the ABI only exposes caller buffers and
# opaque handles. Embed the CRT to avoid depending on an older host VC runtime.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(CTest)
add_subdirectory(native/FrameTrail.Engine)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native/FrameTrail.Capture/CMakeLists.txt")
  add_subdirectory(native/FrameTrail.Capture)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests/FrameTrail.Engine.Tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/FrameTrail.Core.Tests/CMakeLists.txt")
    add_subdirectory(tests/FrameTrail.Core.Tests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/FrameTrail.Audio.Tests/CMakeLists.txt")
    add_subdirectory(tests/FrameTrail.Audio.Tests)
  endif()
endif()
